    │

    │   ├── __init__.py

    │   └── worker.py
    │
    ├── ui/
    │   ├── __init__.py
    │   ├── main_window.py
    │   └── main_window.ui
    │
    ├── controllers/
    │   ├── __init__.py
    │   └── main_controller.py
    │
    └── utils/
        ├── __init__.py
        └── logger.py
